Kotlin
import java.io.BufferedReader
import java.io.File
import java.io.InputStreamReader
import java.net.ServerSocket
import kotlin.concurrent.thread

class RatingProcessorMicroservice {
    private val socket = ServerSocket(1800)

    fun run() {
        println("RatingProcessor pe portul 1800")
        while (true) {
            val c = socket.accept()
            thread {
                val br = BufferedReader(InputStreamReader(c.inputStream))
                val msg = Message.deserialize(br.readLine().toByteArray())
                val (_, ratingStr) = msg.body.split(" ", limit = 2)
                val rating = ratingStr.toIntOrNull()
                if (rating != null && rating in 1..5) {
                    File("ratings.txt").appendText("${msg.sender}\t$rating\n")
                    println("Salvat: ${msg.sender} -> $rating")
                } else {
                    println("Rating invalid: $ratingStr")
                }
                c.close()
            }
        }
    }
}

fun main() = RatingProcessorMicroservice().run()

import java.io.BufferedReader
import java.io.File
import java.io.InputStreamReader
import java.net.ServerSocket

class NumberOfMessagesMicroservice {
    companion object { const val PORT = 1900 }
    private val socket = ServerSocket(PORT)
    private var count = 0

    fun run() {
        println("Counter pe portul $PORT")
        val conn = socket.accept()
        val br = BufferedReader(InputStreamReader(conn.inputStream))
        while (true) {
            val line = br.readLine() ?: break
            count++
            if (Message.deserialize(line.toByteArray()).body == "final") break
        }
        File("statistica.txt").appendText("Licitatie ${System.currentTimeMillis()}: $count mesaje\n")
        println("Numar total: $count")
    }
}

fun main() = NumberOfMessagesMicroservice().run()

import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.ServerSocket
import java.net.Socket
import kotlin.concurrent.thread

class ChatMasterMicroservice {
    private val clients = mutableMapOf<String, Socket>()
    private val socket = ServerSocket(2000)

    fun run() {
        println("ChatMaster pe portul 2000")
        while (true) {
            val c = socket.accept()
            thread {
                val br = BufferedReader(InputStreamReader(c.inputStream))
                val regLine = br.readLine() ?: return@thread
                val regMsg = Message.deserialize(regLine.toByteArray())
                synchronized(clients) { clients[regMsg.sender] = c }

                while (true) {
                    val line = br.readLine() ?: break
                    val m = Message.deserialize(line.toByteArray())
                    val (_, dest, text) = m.body.split(" ", limit = 3)
                    clients[dest]?.getOutputStream()?.write(Message.create(m.sender, text).serialize())
                }
            }
        }
    }
}

fun main() = ChatMasterMicroservice().run()

fun main(args: Array<String>) {
    val name = args[0]
    val sock = Socket("localhost", 2000)
    sock.getOutputStream().write(Message.create(name, "register").serialize())

    thread {
        val br = BufferedReader(InputStreamReader(sock.inputStream))
        while (true) { val l = br.readLine() ?: break; println(Message.deserialize(l.toByteArray())) }
    }

    while (true) {
        val l = readLine() ?: break
        sock.getOutputStream().write(Message.create(name, "to $l").serialize())
    }
}

import com.rabbitmq.client.ConnectionFactory
import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.ServerSocket
import java.sql.DriverManager

class MonitorMicroservice {
    private val socket = ServerSocket(2100)
    private val db = DriverManager.getConnection("jdbc:sqlite:monitor.db")
    private val rabbit = ConnectionFactory().apply { host = "localhost" }.newConnection().createChannel()

    init {
        db.createStatement().execute("CREATE TABLE IF NOT EXISTS events(id INTEGER PRIMARY KEY, sender TEXT, body TEXT)")
        rabbit.queueDeclare("monitor", true, false, false, null)
    }

    fun run() {
        while (true) {
            val c = socket.accept()
            val br = BufferedReader(InputStreamReader(c.inputStream))
            val line = br.readLine() ?: continue
            val m = Message.deserialize(line.toByteArray())
            db.prepareStatement("INSERT INTO events(sender,body) VALUES(?,?)").apply {
                setString(1, m.sender); setString(2, m.body); execute()
            }
            rabbit.basicPublish("", "monitor", null, line.toByteArray())
        }
    }
}

fun main() = MonitorMicroservice().run()
