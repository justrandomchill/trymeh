import socket, requests, json, time
from datetime import date

API_KEY = "brmr2kfrh5rcss140jmg"

def get_symbols():
    r = requests.get(f"https://finnhub.io/api/v1/stock/symbol?exchange=US&token={API_KEY}")
    return [s["symbol"] for s in r.json()[:20]]

def get_news(symbol):
    today = date.today().isoformat()
    r = requests.get(f"https://finnhub.io/api/v1/company-news?symbol={symbol}&from={today}&to={today}&token={API_KEY}")
    return r.json()

srv = socket.socket(); srv.bind(("0.0.0.0", 9999)); srv.listen(1)
print("Astept Spark...")
conn, _ = srv.accept()
print("Spark conectat!")

for sym in get_symbols():
    for news in get_news(sym):
        conn.sendall((json.dumps(news) + "\n").encode())
        time.sleep(3)


SparkClient.kt (NOU)



Kotlin
package com.sd.laborator

import org.apache.spark.SparkConf
import org.apache.spark.streaming.Durations
import org.apache.spark.streaming.api.java.JavaStreamingContext
import org.json.JSONObject

fun main() {
    val conf = SparkConf().setMaster("local[2]").setAppName("News")
    val ssc = JavaStreamingContext(conf, Durations.seconds(5))
    val lines = ssc.socketTextStream("localhost", 9999)

    lines.foreachRDD { rdd ->
        rdd.collect().forEach { line ->
            val j = JSONObject(line)
            if (j.getString("source") == "Yahoo" && j.getString("summary").length > 500) {
                println("URL: ${j.getString("url")}")
                println("DATE: ${j.getLong("datetime")}")
                println("TITLU: ${j.getString("headline")}")
            }
        }
    }
    ssc.start(); ssc.awaitTermination()
}

package com.sd.laborator

import khttp.get
import java.net.ServerSocket
import java.time.LocalDate

fun main() {
    val key = "brmrfu7rh5rcss140ogg"
    val yest = LocalDate.now().minusDays(1).toString()
    val srv = ServerSocket(9999); val conn = srv.accept()
    val syms = get("https://finnhub.io/api/v1/stock/symbol?exchange=US&token=$key").jsonArray
    for (i in 0 until minOf(20, syms.length())) {
        val sym = syms.getJSONObject(i).getString("symbol")
        val news = get("https://finnhub.io/api/v1/company-news?symbol=$sym&from=$yest&to=$yest&token=$key").jsonArray
        for (j in 0 until news.length()) {
            conn.getOutputStream().write((news.getJSONObject(j).toString() + "\n").toByteArray())
            Thread.sleep(3000)
        }
    }
}

from pyspark import SparkContext
from pyspark.streaming import StreamingContext
import json

sc = SparkContext("local[2]", "News"); sc.setLogLevel("ERROR")
ssc = StreamingContext(sc, 5)
lines = ssc.socketTextStream("localhost", 9999)

def process(rdd):
    for line in rdd.collect():
        try:
            j = json.loads(line)
            if j.get("image", "").endswith(".png") and len(j.get("url", "")) < 80:
                print(f"URL={j['url']} DATE={j['datetime']} TITLE={j['headline']}")
        except: pass

lines.foreachRDD(process)
ssc.start(); ssc.awaitTermination()

def process(rdd):
    for line in rdd.collect():
        j = json.loads(line)
        mean = j.get("targetMean", 0); low = j.get("targetLow", 0)
        if low > 0:
            profit = (mean - low) / low * 100
            if profit > 40:
                print(f"SYMBOL={j['symbol']} PROFIT={profit:.2f}%")

    rdd.collect().forEach { line ->
        val j = JSONObject(line)

        val words = text.split(Regex("\\W+"))

@Controller
class SentimentController {
    private val results = mutableListOf<Triple<String, String, String>>()

    @PostMapping("/add")
    @ResponseBody
    fun add(@RequestBody body: Map<String, String>): String {
        results.add(Triple(body["sentiment"]!!, body["title"]!!, body["url"]!!))
        return "OK"
    }

    @GetMapping("/view")
    @ResponseBody
    fun view(): String {
        val rows = results.joinToString("") { (s, t, u) ->
            "<tr><td>$s</td><td>$t</td><td><a href='$u'>link</a></td></tr>"
        }
        return "<html><body><table border=1>$rows</table></body></html>"
    }
}

Diagrama de clase = pentru fiecare microserviciu desenezi clasele importante cu metode publice si dependentele (interfata <-implementeaza- clasa). Subliniaza ISP, DIP.
Apararea SOLID:
SRP: o clasa = o responsabilitate (filtru, cache, log etc.)
OCP: adaugi un Logger fara sa modifici BeerAdderService (decorator)
LSP: orice IBeerAdder poate fi inlocuit cu LoggingBeerAdderService
ISP: am spart IBeerService in 4 interfete mici
DIP: controller-ul depinde de IBeerAdder, nu de implementare
Daca pica o varianta neacoperita: combini ce ai mai sus (pattern-uri: cache pe disc + filtru + dirijor + replicator + observator + decorator + gateway).
