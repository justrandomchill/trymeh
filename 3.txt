package com.sd.laborator.services

import com.sd.laborator.interfaces.*
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.stereotype.Service

@Service
class WeatherAPIService : IWeatherAPI {
    @Autowired private lateinit var loc: LocationSearchInterface
    @Autowired private lateinit var fc: WeatherForecastInterface

    override fun getForecast(city: String): String {
        val id = loc.getLocationId(city)
        if (id == -1) return "Nu s-a gasit orasul"
        return fc.getForecastData(id).toString()
    }
}

package com.sd.laborator.services

import com.sd.laborator.interfaces.*
import org.springframework.stereotype.Service

@Service
class FilterService : IFilterService {
    override fun filter(raw: String): CityTemp {
        val city = Regex("location=([^,]+)").find(raw)!!.groupValues[1]
        val min = Regex("minTemp=(-?\\d+)").find(raw)!!.groupValues[1].toInt()
        val max = Regex("maxTemp=(-?\\d+)").find(raw)!!.groupValues[1].toInt()
        return CityTemp(city, min, max)
    }
}

package com.sd.laborator.services

import com.sd.laborator.interfaces.*
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.stereotype.Service

@Service
class DirijorService {
    @Autowired private lateinit var api: IWeatherAPI
    @Autowired private lateinit var filter: IFilterService
    @Autowired private lateinit var writer: IFileWriter

    fun process(city: String) {
        val raw = api.getForecast(city)
        val filtered = filter.filter(raw)
        writer.write(filtered)
    }
}

package com.sd.laborator.services

import org.springframework.stereotype.Service
import kotlin.math.atan
import kotlin.math.PI

@Service
class RegressionService {
    // y = ax + b, returneaza unghiul dreptei in grade
    fun angle(temps: List<Double>): Double {
        val n = temps.size
        val x = (1..n).map { it.toDouble() }
        val mx = x.average(); val my = temps.average()
        val num = x.zip(temps).sumOf { (xi, yi) -> (xi - mx) * (yi - my) }
        val den = x.sumOf { (it - mx) * (it - mx) }
        val a = num / den
        return atan(a) * 180.0 / PI
    }
}

package com.sd.laborator.services

import org.springframework.stereotype.Service
import java.net.ServerSocket
import kotlin.concurrent.thread

@Service
class ReplicatorService(private val api: com.sd.laborator.interfaces.IWeatherAPI) {
    fun replicate(n: Int, basePort: Int = 5000) {
        repeat(n) { i ->
            thread {
                val ss = ServerSocket(basePort + i)
                println("Replica ${i+1} pe portul ${basePort + i}")
                while (true) {
                    val cl = ss.accept()
                    val city = cl.getInputStream().bufferedReader().readLine()
                    cl.getOutputStream().write((api.getForecast(city) + "\n").toByteArray())
                    cl.close()
                }
            }
        }
    }
}


weather_gui.py (NOU)



Python
import sys, socket
from PyQt5.QtWidgets import QApplication, QWidget, QLineEdit, QPushButton, QTextEdit, QVBoxLayout

class W(QWidget):
    def __init__(self):
        super().__init__()
        self.line = QLineEdit(); self.btn = QPushButton("Forecast"); self.txt = QTextEdit()
        lay = QVBoxLayout(); lay.addWidget(self.line); lay.addWidget(self.btn); lay.addWidget(self.txt)
        self.setLayout(lay); self.btn.clicked.connect(self.go)

    def go(self):
        s = socket.socket(); s.connect(("localhost", 5000))
        s.sendall((self.line.text() + "\n").encode())
        self.txt.setText(s.recv(4096).decode()); s.close()

app = QApplication(sys.argv); w = W(); w.show(); sys.exit(app.exec_())
