package com.sd.laborator

import java.io.BufferedReader
import java.io.File
import java.io.InputStreamReader
import java.net.ServerSocket
import java.net.Socket
import kotlin.concurrent.thread

class PortFilterMicroservice {
    companion object {
        const val FILTER_PORT = 1400
        const val MIN_PORT = 1000
        const val MAX_PORT = 2000
    }

    fun run() {
        val ss = ServerSocket(FILTER_PORT)
        println("PortFilter pe portul $FILTER_PORT, accepta $MIN_PORT-$MAX_PORT")
        while (true) {
            val client = ss.accept()
            thread {
                val downstream = Socket("messagemanager", 1500)
                val br = BufferedReader(InputStreamReader(client.inputStream))
                while (true) {
                    val line = br.readLine() ?: break
                    if (client.port in MIN_PORT..MAX_PORT) {
                        File("results.txt").appendText("[ACCEPT ${client.port}] $line\n")
                        downstream.getOutputStream().write((line + "\n").toByteArray())
                    } else {
                        File("results.txt").appendText("[DROP ${client.port}] $line\n")
                    }
                }
                client.close(); downstream.close()
            }
        }
    }
}
fun main() = PortFilterMicroservice().run()

package com.sd.laborator

import kotlinx.coroutines.*
import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.ServerSocket

data class UserInfo(val addr: String, val port: Int)

class ActiveUsersMicroservice {
    private val users = mutableMapOf<String, UserInfo>()
    private val socket = ServerSocket(1500)

    fun run() = runBlocking {
        println("ActiveUsers pe 1500")
        while (true) {
            val c = socket.accept()
            launch(Dispatchers.IO) {
                val br = BufferedReader(InputStreamReader(c.inputStream))
                while (true) {
                    val line = br.readLine() ?: break
                    val parts = line.split(" ")
                    when (parts[0]) {
                        "register" -> {
                            synchronized(users) { users[parts[1]] = UserInfo(c.inetAddress.hostAddress, parts[2].toInt()) }
                            c.getOutputStream().write("OK\n".toByteArray())
                        }
                        "lookup" -> {
                            val u = users[parts[1]]
                            c.getOutputStream().write(((u?.let { "${it.addr}:${it.port}" } ?: "NULL") + "\n").toByteArray())
                        }
                        "unregister" -> { synchronized(users) { users.remove(parts[1]) } }
                    }
                }
            }
        }
    }
}
fun main() = ActiveUsersMicroservice().run()

package com.sd.laborator

import kotlinx.coroutines.*
import java.io.File
import java.net.ServerSocket

class FileTransferMicroservice {
    fun run() = runBlocking {
        val ss = ServerSocket(1700)
        println("FileTransfer pe 1700")
        while (true) {
            val c = ss.accept()
            launch(Dispatchers.IO) {
                val dis = java.io.DataInputStream(c.inputStream)
                val cmd = dis.readUTF()  // "SEND <filename> <size>"
                if (cmd.startsWith("SEND")) {
                    val (_, fname, sizeStr) = cmd.split(" ")
                    val size = sizeStr.toInt()
                    val buf = ByteArray(size)
                    dis.readFully(buf)
                    File("received_$fname").writeBytes(buf)
                    println("Primit fisier: $fname ($size bytes)")
                }
            }
        }
    }
}
fun main() = FileTransferMicroservice().run()

package com.sd.laborator

import kotlinx.coroutines.*
import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.ServerSocket
import java.net.Socket

class MulticastMicroservice {
    private val groups = mutableMapOf<String, MutableList<Socket>>()
    fun run() = runBlocking {
        val ss = ServerSocket(1800)
        while (true) {
            val c = ss.accept()
            launch(Dispatchers.IO) {
                val br = BufferedReader(InputStreamReader(c.inputStream))
                while (true) {
                    val line = br.readLine() ?: break
                    val parts = line.split(" ", limit = 3)
                    when (parts[0]) {
                        "join" -> synchronized(groups) {
                            groups.getOrPut(parts[1]) { mutableListOf() }.add(c)
                        }
                        "send" -> {
                            groups[parts[1]]?.forEach { peer ->
                                peer.getOutputStream().write(("[${parts[1]}] ${parts[2]}\n").toByteArray())
                            }
                        }
                    }
                }
            }
        }
    }
}
fun main() = MulticastMicroservice().run()

package com.sd.laborator

import kotlinx.coroutines.*
import java.io.BufferedReader
import java.io.File
import java.io.InputStreamReader
import java.net.ServerSocket
import java.net.Socket

class CensorMicroservice {
    private val dict = File("dictionary.txt").readLines().toSet()

    private fun censor(msg: String): String {
        var out = msg
        for (w in dict) out = out.replace(w, "x".repeat(w.length), ignoreCase = true)
        return out
    }

    fun run() = runBlocking {
        val ss = ServerSocket(1900)
        while (true) {
            val client = ss.accept()
            launch(Dispatchers.IO) {
                val down = Socket("localhost", 1500)
                val br = BufferedReader(InputStreamReader(client.inputStream))
                while (true) {
                    val line = br.readLine() ?: break
                    down.getOutputStream().write((censor(line) + "\n").toByteArray())
                }
            }
        }
    }
}
fun main() = CensorMicroservice().run()

package com.sd.laborator

import com.sd.laborator.interfaces.*
import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.ServerSocket
import kotlin.concurrent.thread

class RegistryMicroservice : IRegistry {
    private val services = mutableMapOf<String, String>()
    private val observers = mutableListOf<IObserver>()

    override fun register(n: String, a: String) {
        synchronized(services) { services[n] = a }
        observers.forEach { it.onRegistered(n) }
    }
    override fun unregister(n: String) {
        synchronized(services) { services.remove(n) }
        observers.forEach { it.onUnregistered(n) }
    }
    override fun subscribe(o: IObserver) { observers.add(o) }

    fun run() {
        val ss = ServerSocket(1500)
        while (true) {
            val c = ss.accept()
            thread {
                val br = BufferedReader(InputStreamReader(c.inputStream))
                while (true) {
                    val line = br.readLine() ?: break
                    val parts = line.split(" ")
                    when (parts[0]) {
                        "register" -> register(parts[1], "${c.inetAddress.hostAddress}:${parts[2]}")
                        "unregister" -> unregister(parts[1])
                        "list" -> c.getOutputStream().write((services.toString() + "\n").toByteArray())
                    }
                }
                services.keys.filter { services[it]?.startsWith(c.inetAddress.hostAddress) == true }
                    .forEach { unregister(it) }
            }
        }
    }
}
fun main() = RegistryMicroservice().run()

package com.sd.laborator

import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.ServerSocket
import java.sql.DriverManager
import kotlin.concurrent.thread

class DbWriterMicroservice {
    private val db = DriverManager.getConnection("jdbc:sqlite:messages.db")
    init {
        db.createStatement().execute("CREATE TABLE IF NOT EXISTS msgs(id INTEGER PRIMARY KEY, port INT, body TEXT)")
    }
    fun run() {
        val ss = ServerSocket(2200)
        while (true) {
            val c = ss.accept()
            thread {
                val br = BufferedReader(InputStreamReader(c.inputStream))
                while (true) {
                    val line = br.readLine() ?: break
                    val ps = db.prepareStatement("INSERT INTO msgs(port,body) VALUES(?,?)")
                    ps.setInt(1, c.port); ps.setString(2, line); ps.execute()
                }
            }
        }
    }
}
fun main() = DbWriterMicroservice().run()
