src/main/kotlin/com/sd/laborator/services/CacheService.kt (NOU - cache cu TTL 30 min)



Kotlin
package com.sd.laborator.services

import com.sd.laborator.interfaces.ICacheService
import org.springframework.stereotype.Service
import java.util.concurrent.ConcurrentHashMap

@Service
class CacheService : ICacheService {
    private data class Entry(val value: String, val timestamp: Long)
    private val store = ConcurrentHashMap<String, Entry>()
    private val ttlMs = 30 * 60 * 1000L  // 30 minute

    override fun get(key: String): String? {
        val e = store[key] ?: return null
        if (System.currentTimeMillis() - e.timestamp > ttlMs) {
            store.remove(key); return null
        }
        return e.value
    }

    override fun put(key: String, value: String) {
        store[key] = Entry(value, System.currentTimeMillis())
    }
}


src/main/kotlin/com/sd/laborator/controllers/AgendaController.kt (NOU - poarta API)



Kotlin
package com.sd.laborator.controllers

import com.sd.laborator.interfaces.IAgendaService
import com.sd.laborator.interfaces.ICacheService
import com.sd.laborator.pojo.Person
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.web.bind.annotation.*

@RestController
@RequestMapping("/agenda")
class AgendaController {
    @Autowired private lateinit var agenda: IAgendaService
    @Autowired private lateinit var cache: ICacheService

    @PostMapping("/add")
    fun add(@RequestBody p: Person): String = agenda.addPerson(p)

    @GetMapping("/get/{name}")
    fun get(@PathVariable name: String): String {
        val hit = cache.get(name)
        if (hit != null) return "[CACHE HIT] $hit"
        val person = agenda.getPerson(name) ?: return "Nu exista"
        val result = person.toString()
        cache.put(name, result)
        return "[CACHE MISS] $result"
    }
}

Kotlin
package com.sd.laborator.services

import com.sd.laborator.interfaces.ICacheService
import org.springframework.stereotype.Service

@Service
class GlobalCacheService(private val replicas: List<CacheService>) : ICacheService {
    override fun get(key: String): String? {
        for (r in replicas) {
            val hit = r.get(key)
            if (hit != null) return hit
        }
        return null
    }
    override fun put(key: String, value: String) {
        replicas.forEach { it.put(key, value) }  // consistenta tare
    }
}

import com.sd.laborator.interfaces.IExpenseService
import com.sd.laborator.services.GlobalCacheService
import com.sd.laborator.pojo.Expense
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.web.bind.annotation.*

@RestController
@RequestMapping("/expense")
class ExpenseController {
    @Autowired private lateinit var svc: IExpenseService
    @Autowired private lateinit var cache: GlobalCacheService

    @PostMapping("/add")
    fun add(@RequestBody e: Expense): String { svc.add(e); return "OK" }

    @GetMapping("/get/{m}")
    fun get(@PathVariable m: String): String {
        cache.get(m)?.let { return "[CACHE] $it" }
        val res = svc.getByMember(m).toString()
        cache.put(m, res); return "[DB] $res"
    }
}

package com.sd.laborator.services

import com.rabbitmq.client.ConnectionFactory
import com.sd.laborator.interfaces.IChelner
import org.springframework.beans.factory.annotation.Qualifier
import org.springframework.stereotype.Service

@Service
class Chelner(
    val id: Int,
    @Qualifier("rabbitFactory") private val factory: ConnectionFactory
) : IChelner {
    override fun preiaComanda(): String {
        val comanda = "Chelner-$id: pizza ${System.currentTimeMillis()}"
        factory.newConnection().use { conn ->
            conn.createChannel().use { ch ->
                ch.queueDeclare("comenzi", true, false, false, null)
                ch.basicPublish("", "comenzi", null, comanda.toByteArray())
            }
        }
        return comanda
    }
}

package com.sd.laborator.services

import com.rabbitmq.client.*
import com.sd.laborator.interfaces.IBucatar
import org.springframework.beans.factory.annotation.Qualifier
import org.springframework.stereotype.Service

@Service
class Bucatar(
    val id: Int,
    @Qualifier("rabbitFactory") private val factory: ConnectionFactory
) : IBucatar {
    override fun gateste(comanda: String) { println("Bucatar-$id gateste: $comanda") }

    fun start() {
        val conn = factory.newConnection()
        val ch = conn.createChannel()
        ch.queueDeclare("comenzi", true, false, false, null)
        ch.basicQos(1)
        val cb = DeliverCallback { _, d -> gateste(String(d.body)); ch.basicAck(d.envelope.deliveryTag, false) }
        ch.basicConsume("comenzi", false, cb) { _ -> }
    }
}

package com.sd.laborator.config

import com.rabbitmq.client.ConnectionFactory
import com.sd.laborator.services.Bucatar
import com.sd.laborator.services.Chelner
import org.springframework.context.annotation.Bean
import org.springframework.context.annotation.Configuration

@Configuration
class AppConfig {
    @Bean("rabbitFactory") fun rabbit() = ConnectionFactory().apply { host = "localhost" }
    @Bean fun chelner1(f: ConnectionFactory) = Chelner(1, f)
    @Bean fun chelner2(f: ConnectionFactory) = Chelner(2, f)
    @Bean fun bucatar1(f: ConnectionFactory) = Bucatar(1, f).also { it.start() }
    @Bean fun bucatar2(f: ConnectionFactory) = Bucatar(2, f).also { it.start() }
    @Bean fun bucatar3(f: ConnectionFactory) = Bucatar(3, f).also { it.start() }
}
