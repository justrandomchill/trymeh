package com.sd.laborator.presentation.controllers

import com.sd.laborator.business.interfaces.*
import com.sd.laborator.models.Beer
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.web.bind.annotation.*

@RestController
@RequestMapping("/api/beer")
class BeerApiGateway {
    @Autowired private lateinit var adder: IBeerAdder
    @Autowired private lateinit var reader: IBeerReader
    @Autowired private lateinit var updater: IBeerUpdater
    @Autowired private lateinit var deleter: IBeerDeleter

    @PostMapping("/add") fun add(@RequestBody b: Beer) = adder.addBeer(b).let { "OK" }
    @GetMapping("/all") fun all() = reader.getBeers()
    @GetMapping("/name/{n}") fun byName(@PathVariable n: String) = reader.getBeerByName(n)
    @GetMapping("/price/{p}") fun byPrice(@PathVariable p: Float) = reader.getBeerByPrice(p)
    @PutMapping("/update") fun upd(@RequestBody b: Beer) = updater.updateBeer(b).let { "OK" }
    @DeleteMapping("/del/{n}") fun del(@PathVariable n: String) = deleter.deleteBeer(n).let { "OK" }
}

package com.sd.laborator.business.services

import com.sd.laborator.business.interfaces.IBeerAdder
import com.sd.laborator.models.Beer
import org.springframework.beans.factory.annotation.Value
import java.io.File
import java.time.LocalDateTime

class LoggingBeerAdderService(
    private val wrapped: IBeerAdder,
    @Value("\${logging.crudLogs}") private val logFile: String
) : IBeerAdder {
    override fun addBeer(beer: Beer) {
        File(logFile).appendText("""{"op":"ADD","beer":"${beer.beerName}","ts":"${LocalDateTime.now()}"}""" + "\n")
        wrapped.addBeer(beer)
    }
}

package com.sd.laborator.business.services

import org.springframework.stereotype.Service
import java.io.File
import java.time.Instant

@Service
class FileCacheService {
    private val file = File("cache.txt")  // append-only
    private val ttl = 30 * 60L  // 30 minute

    fun put(key: String, value: String) {
        file.appendText("${Instant.now().epochSecond}\t$key\t$value\n")
    }

    fun get(key: String): String? {
        if (!file.exists()) return null
        val now = Instant.now().epochSecond
        return file.readLines().reversed()
            .firstOrNull {
                val (ts, k, _) = it.split("\t", limit = 3)
                k == key && now - ts.toLong() < ttl
            }?.split("\t", limit = 3)?.get(2)
    }
}

package com.sd.laborator.kafka

import com.sd.laborator.business.interfaces.IBeerAdder
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.kafka.annotation.KafkaListener
import org.springframework.kafka.core.KafkaTemplate
import org.springframework.stereotype.Component

@Component
class AddListener {
    @Autowired private lateinit var adder: IBeerAdder
    @Autowired private lateinit var tpl: KafkaTemplate<String, String>

    @KafkaListener(topics = ["beer-add-req"])
    fun onAdd(msg: String) {
        val (n, p) = msg.split(",")
        adder.addBeer(com.sd.laborator.models.Beer(0, n, p.toFloat()))
        tpl.send("beer-resp", "ADDED $n")
    }
}
