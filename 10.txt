import pika, datetime

class ErrorProcessor:
    def __init__(self):
        self.conn = pika.BlockingConnection(pika.ConnectionParameters("localhost"))
        self.ch = self.conn.channel()
        self.ch.queue_declare(queue="errors")
        self.stats = {}

    def on_msg(self, ch, method, props, body):
        err_type = body.decode().split("|")[0]
        self.stats[err_type] = self.stats.get(err_type, 0) + 1
        with open("errors.txt", "a") as f:
            f.write(f"{datetime.datetime.now()}\t{body.decode()}\n")
        print("Stats:", self.stats)

    def run(self):
        self.ch.basic_consume(queue="errors", on_message_callback=self.on_msg, auto_ack=True)
        self.ch.start_consuming()

if __name__ == "__main__":
    ErrorProcessor().run()

import pika, datetime

class LoggerProcessor:
    QUEUES = ["topic_oferte", "topic_rezultat", "topic_oferte_procesate", "topic_notificare_procesor_mesaje"]

    def __init__(self):
        self.conn = pika.BlockingConnection(pika.ConnectionParameters("localhost"))
        self.ch = self.conn.channel()
        for q in self.QUEUES:
            self.ch.exchange_declare(exchange=f"{q}_fanout", exchange_type="fanout")
            res = self.ch.queue_declare("", exclusive=True)
            self.ch.queue_bind(exchange=f"{q}_fanout", queue=res.method.queue)
            self.ch.basic_consume(queue=res.method.queue,
                on_message_callback=lambda c,m,p,b,qn=q: self.log(qn, b), auto_ack=True)

    def log(self, queue, body):
        with open("application_log.log", "a") as f:
            f.write(f"{datetime.datetime.now()}\t[{queue}]\t{body}\n")
        print(f"[{queue}] {body}")

    def run(self): self.ch.start_consuming()

if __name__ == "__main__":
    LoggerProcessor().run()

import pika

class ChatMaster:
    def __init__(self):
        self.conn = pika.BlockingConnection(pika.ConnectionParameters("localhost"))
        self.ch = self.conn.channel()
        self.ch.queue_declare(queue="chat_inbox")

    def on_msg(self, ch, m, p, body):
        sender, recv, text = body.decode().split("|")
        ch.queue_declare(queue=f"chat_{recv}")
        ch.basic_publish(exchange="", routing_key=f"chat_{recv}", body=f"{sender}: {text}".encode())

    def run(self):
        self.ch.basic_consume(queue="chat_inbox", on_message_callback=self.on_msg, auto_ack=True)
        self.ch.start_consuming()

if __name__ == "__main__":
    ChatMaster().run()

import pika, sys, threading

name = sys.argv[1]
conn = pika.BlockingConnection(pika.ConnectionParameters("localhost"))
ch = conn.channel()
ch.queue_declare(queue=f"chat_{name}")

def listen():
    for m, p, body in ch.consume(f"chat_{name}", auto_ack=True):
        print(body.decode())

threading.Thread(target=listen, daemon=True).start()

while True:
    line = input("to|text: ")
    to, text = line.split("|", 1)
    ch.basic_publish(exchange="", routing_key="chat_inbox", body=f"{name}|{to}|{text}".encode())

import pika

class RatingProcessor:
    def __init__(self):
        self.conn = pika.BlockingConnection(pika.ConnectionParameters("localhost"))
        self.ch = self.conn.channel()
        self.ch.queue_declare(queue="ratings")

    def on_msg(self, ch, m, p, body):
        user, r = body.decode().split("|")
        r = int(r)
        if 1 <= r <= 5:
            with open("ratings.txt", "a") as f: f.write(f"{user}\t{r}\n")
            print(f"Salvat: {user} -> {r}")

    def run(self):
        self.ch.basic_consume(queue="ratings", on_message_callback=self.on_msg, auto_ack=True)
        self.ch.start_consuming()

if __name__ == "__main__":
    RatingProcessor().run()

import pika

class MessageCounter:
    def __init__(self):
        self.count = 0
        self.conn = pika.BlockingConnection(pika.ConnectionParameters("localhost"))
        self.ch = self.conn.channel()
        self.ch.exchange_declare("topic_oferte_fanout", "fanout")
        res = self.ch.queue_declare("", exclusive=True)
        self.ch.queue_bind(exchange="topic_oferte_fanout", queue=res.method.queue)
        self.queue_name = res.method.queue

    def on_msg(self, ch, m, p, body):
        self.count += 1
        if body == b"final":
            with open("nr_mesaje.txt", "a") as f: f.write(f"Licitatie: {self.count} mesaje\n")
            print(f"Total mesaje: {self.count}")
            self.ch.stop_consuming()

    def run(self):
        self.ch.basic_consume(queue=self.queue_name, on_message_callback=self.on_msg, auto_ack=True)
        self.ch.start_consuming()

if __name__ == "__main__":
    MessageCounter().run()
