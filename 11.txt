package com.sd.laborator
import jakarta.inject.Singleton

@Singleton
class SumOfSquaresService {
    fun compute(a: List<Int>, n: Int): Long = a.take(n).sumOf { (it.toLong() * it.toLong()) }
}

EratosteneFunction.kt -> SumFunction.kt (MODIFICA)



Kotlin
@FunctionBean("sumOfSquares")
class SumFunction : FunctionInitializer(), Function<SumRequest, SumResponse> {
    @Inject private lateinit var svc: SumOfSquaresService

    override fun apply(req: SumRequest): SumResponse {
        val r = SumResponse(); r.bn = svc.compute(req.a, req.n); r.message = "OK"; return r
    }
}

}

11.5 Automat (state machine) cu functii serverless + cozi
Enunt
Cu functii serverless, sa se implementeze un automat (state machine, ex. 5 stari). Fiecare stare = o functie serverless care citeste dintr-o coada verificata de toate; un serviciu REST preia datele de la utilizator si le pune in coada. Variante: Subiect 1 = cozi RabbitMQ, Subiect 2 = TCP.

AutomatService.kt (NOU)



Kotlin
@Singleton
class AutomatService {
    // 5 stari: S0 -> S1 -> S2 -> S3 -> S4 (final)
    val transitions = mapOf(
        "S0" to mapOf(0 to "S1", 1 to "S2"),
        "S1" to mapOf(0 to "S3", 1 to "S0"),
        "S2" to mapOf(0 to "S0", 1 to "S4"),
        "S3" to mapOf(0 to "S4", 1 to "S2"),
        "S4" to mapOf(0 to "S4", 1 to "S4")
    )
    var current = "S0"
    fun step(input: Int): String { current = transitions[current]?.get(input) ?: current; return current }
}


AutomatFunction.kt (NOU)



Kotlin
@FunctionBean("automat")
class AutomatFunction : FunctionInitializer(), Function<Map<String,Any>, Map<String,String>> {
    @Inject private lateinit var svc: AutomatService
    override fun apply(req: Map<String,Any>): Map<String,String> {
        val s = svc.step((req["input"] as Number).toInt())
        return mapOf("state" to s)
    }
}


AutomatController.kt (NOU)



Kotlin
@Controller("/")
class AutomatController {
    @Inject private lateinit var fn: AutomatFunction
    @Post fun receive(@Body req: Map<String,Any>) = fn.apply(req)
}


RabbitListener.kt (NOU)



Kotlin
@Singleton
class RabbitListener(private val fn: AutomatFunction) {
    @PostConstruct fun start() {
        val conn = ConnectionFactory().apply { host = "localhost" }.newConnection()
        val ch = conn.createChannel(); ch.queueDeclare("automat-in", true, false, false, null)
        ch.basicConsume("automat-in", true, { _, d ->
            val input = String(d.body).toInt()
            println("Stare: ${fn.apply(mapOf("input" to input))}")
        }, { _ -> })
    }
}
